```python
class HybridMiniSwin2_5D_ResNet_CSRF(nn.Module):
    def __init__(self, in_channels=15, num_classes=1):
        super().__init__()
        
        # Encoder
        self.encoder1 = nn.Sequential(...)  # 256×256×32
        self.encoder2 = nn.Sequential(...)  # 128×128×64
        self.encoder3 = nn.Sequential(...)  # 64×64×128
        self.encoder4 = nn.Sequential(...)  # 32×32×256
        
        # Bottleneck with CSRF
        self.bottleneck = ResNetBottleneck(256)
        self.csrf = ChannelSpatialRefinement(256)
        
        # Decoder
        self.decoder1 = nn.Sequential(...)  # 64×64×128
        self.decoder2 = nn.Sequential(...)  # 128×128×64
        self.decoder3 = nn.Sequential(...)  # 256×256×32
        
        # Final layer
        self.final = nn.Conv2d(32, num_classes, kernel_size=1)
        self.sigmoid = nn.Sigmoid()
    
    def forward(self, x):
        # Encoder with skip connections
        e1 = self.encoder1(x)
        e2 = self.encoder2(e1)
        e3 = self.encoder3(e2)
        e4 = self.encoder4(e3)
        
        # Bottleneck
        b = self.bottleneck(e4)
        b = self.csrf(b)
        
        # Decoder with skip connections
        d1 = self.decoder1(torch.cat([b, e3], dim=1))
        d2 = self.decoder2(torch.cat([d1, e2], dim=1))
        d3 = self.decoder3(torch.cat([d2, e1], dim=1))
        
        # Output
        out = self.final(d3)
        out = self.sigmoid(out)
        return out
```

Training Loop Structure:

```python
def main():
    # 1. Setup
    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    model = HybridMiniSwin2_5D_ResNet_CSRF().to(device)
    
    # 2. Load MAE pre-trained weights
    mae_checkpoint = load_mae_checkpoint('path/to/mae.pth')
    model.load_state_dict(mae_checkpoint, strict=False)
    
    # 3. Setup training
    optimizer = torch.optim.AdamW(model.parameters(), lr=1e-4, weight_decay=1e-4)
    scheduler = ReduceLROnPlateau(optimizer, mode='max', patience=5)
    loss_fn = FocalTverskyLoss(alpha=0.7, beta=0.3, gamma=1.33)
    
    # 4. Training loop
    best_dice = 0.0
    patience_counter = 0
    
    for epoch in range(100):
        # Train
        train_loss = train_epoch(model, train_loader, optimizer, loss_fn, device)
        
        # Validate
        val_metrics = validate_epoch(model, val_loader, loss_fn, device)
        val_dice = val_metrics['dice']
        
        # Logging
        print(f"Epoch {epoch+1}: Train Loss = {train_loss:.4f}, Val Dice = {val_dice:.4f}")
        
        # Learning rate scheduling
        scheduler.step(val_dice)
        
        # Early stopping
        if val_dice > best_dice:
            best_dice = val_dice
            save_checkpoint(model, optimizer, epoch, val_metrics, 'best_model.pth')
            patience_counter = 0
        else:
            patience_counter += 1
            if patience_counter >= 10:
                print("Early stopping triggered")
                break
```

```python
def load_checkpoint(path):
    """Load checkpoint with error handling"""
    try:
        checkpoint = torch.load(path, map_location='cpu')
        assert 'model_state_dict' in checkpoint, "Invalid checkpoint"
        assert 'epoch' in checkpoint, "Missing epoch info"
        return checkpoint
    except FileNotFoundError:
        raise FileNotFoundError(f"Checkpoint not found: {path}")
    except Exception as e:
        raise RuntimeError(f"Failed to load checkpoint: {e}")
```

```python
    """Forward pass with memory management"""
    try:
        with torch.cuda.amp.autocast():  # Mixed precision
            output = model(batch.to(device))
        return output
    except RuntimeError as e:
        if "out of memory" in str(e):
            torch.cuda.empty_cache()
```

```python
class MSLesionDataset(Dataset):
    def __init__(self, data_dir, split='train', transform=None):
        """
        Args:
            data_dir: Root directory containing patient folders
            split: 'train' or 'val'
            transform: Augmentation transforms
        """
        self.data_dir = data_dir
        self.split = split
        self.transform = transform
        self.patient_list = self._load_patient_list()
    
    def __len__(self):
        return len(self.patient_list)
    
    def __getitem__(self, idx):
        patient_id = self.patient_list[idx]
        
        # Load volumes
        flair = self._load_nifti(f"{patient_id}_FLAIR.nii.gz")
        t1 = self._load_nifti(f"{patient_id}_T1.nii.gz")
        t2 = self._load_nifti(f"{patient_id}_T2.nii.gz")
        mask = self._load_nifti(f"{patient_id}_mask.nii.gz")
        
        # Stack modalities
        volume = np.stack([flair, t1, t2], axis=-1)  # (H, W, D, 3)
        
        # Apply transforms
        if self.transform:
            volume, mask = self.transform(volume, mask)
        
        # Convert to tensor
        volume = torch.from_numpy(volume).float()
        mask = torch.from_numpy(mask).float()
        
        return volume, mask
```

Function: create_dataloaders()

```python
def create_dataloaders(data_dir, batch_size=4, num_workers=4):
    """Create train and validation dataloaders"""
    
    # Training transforms
    train_transform = Compose([
        RandomRotation(degrees=15),
        RandomFlip(p=0.5),
        RandomIntensityShift(std=0.1),
        Normalize(mean=0.0, std=1.0)
    ])
    
    # Validation transforms (no augmentation)
    val_transform = Compose([
        Normalize(mean=0.0, std=1.0)
    ])
    
    # Create datasets
    train_dataset = MSLesionDataset(data_dir, split='train', transform=train_transform)
    val_dataset = MSLesionDataset(data_dir, split='val', transform=val_transform)
    
    # Create dataloaders
    train_loader = DataLoader(
        train_dataset,
        batch_size=batch_size,
        shuffle=True,
        num_workers=num_workers,
        pin_memory=True
    )
    
    val_loader = DataLoader(
        val_dataset,
        batch_size=batch_size,
        shuffle=False,
        num_workers=num_workers,
        pin_memory=True
    )
    
    return train_loader, val_loader
```

4.2 METRICS COMPUTATION
------------------------

Function: compute_dice_score()

```python
def compute_dice_score(pred, target, threshold=0.5, smooth=1e-6):
    """
    Compute Dice Score (F1 Score for segmentation)
    
    Args:
        pred: Predicted probabilities (B, 1, H, W)
        target: Ground truth masks (B, 1, H, W)
        threshold: Threshold for binarization
        smooth: Smoothing factor to avoid division by zero
    
    Returns:
        Dice score (float)
    """
    pred_binary = (pred > threshold).float()
    target_binary = target.float()
    
    intersection = (pred_binary * target_binary).sum()
    union = pred_binary.sum() + target_binary.sum()
    
    dice = (2.0 * intersection + smooth) / (union + smooth)
    return dice.item()
```

Function: compute_metrics()

```python
def compute_metrics(pred, target, threshold=0.5):
    """
    Compute comprehensive segmentation metrics
    
    Returns:
        Dictionary with Dice, Recall, Precision, F1, IoU
    """
    pred_binary = (pred > threshold).float()
    target_binary = target.float()
    
    # True Positives, False Positives, False Negatives
    TP = ((pred_binary == 1) & (target_binary == 1)).sum().item()
    FP = ((pred_binary == 1) & (target_binary == 0)).sum().item()
    FN = ((pred_binary == 0) & (target_binary == 1)).sum().item()
    TN = ((pred_binary == 0) & (target_binary == 0)).sum().item()
    
    # Compute metrics
    dice = (2 * TP) / (2 * TP + FP + FN + 1e-6)
    recall = TP / (TP + FN + 1e-6)
    precision = TP / (TP + FP + 1e-6)
    f1 = 2 * (precision * recall) / (precision + recall + 1e-6)
    iou = TP / (TP + FP + FN + 1e-6)
    specificity = TN / (TN + FP + 1e-6)
    
    return {
        'dice': dice,
        'recall': recall,
        'precision': precision,
        'f1': f1,
        'iou': iou,
        'specificity': specificity
    }
```

```python
class FocalTverskyLoss(nn.Module):
    def __init__(self, alpha=0.7, beta=0.3, gamma=1.33, smooth=1e-6):
        """
        Focal Tversky Loss
        
        Args:
            alpha: Weight for false negatives (higher = emphasize recall)
            beta: Weight for false positives
            gamma: Focal parameter (higher = focus on hard examples)
            smooth: Smoothing factor
        """
        super(FocalTverskyLoss, self).__init__()
        self.alpha = alpha
        self.beta = beta
        self.gamma = gamma
        self.smooth = smooth
    
    def forward(self, pred, target):
        """
        Args:
            pred: Predicted probabilities (B, 1, H, W)
            target: Ground truth masks (B, 1, H, W)
        
        Returns:
            Loss value (scalar)
        """
        # Flatten
        pred = pred.view(-1)
        target = target.view(-1)
        
        # True Positives, False Positives, False Negatives
        TP = (pred * target).sum()
        FP = (pred * (1 - target)).sum()
        FN = ((1 - pred) * target).sum()
        
        # Tversky Index
        tversky_index = (TP + self.smooth) / (TP + self.alpha * FN + self.beta * FP + self.smooth)
        
        # Focal Tversky Loss
        focal_tversky = (1 - tversky_index) ** self.gamma
        
        return focal_tversky
```

Alternative Loss: DiceLoss

```python
class DiceLoss(nn.Module):
    def __init__(self, smooth=1e-6):
        super(DiceLoss, self).__init__()
        self.smooth = smooth
    
    def forward(self, pred, target):
        pred = pred.view(-1)
        target = target.view(-1)
        
        intersection = (pred * target).sum()
        dice = (2.0 * intersection + self.smooth) / (pred.sum() + target.sum() + self.smooth)
        
        return 1 - dice
```

Function: train_epoch()

```python
def train_epoch(model, dataloader, optimizer, loss_fn, device, scaler=None):
    """
    Train for one epoch
    
    Args:
        model: PyTorch model
        dataloader: Training dataloader
        optimizer: Optimizer (AdamW)
        loss_fn: Loss function (FocalTverskyLoss)
        device: 'cuda' or 'cpu'
        scaler: GradScaler for mixed precision
    
    Returns:
        Average training loss
    """
    model.train()
    total_loss = 0.0
    
    for batch_idx, (volumes, masks) in enumerate(tqdm(dataloader)):
        volumes = volumes.to(device)
        masks = masks.to(device)
        
        optimizer.zero_grad()
        
        # Mixed precision training
        if scaler:
            with torch.cuda.amp.autocast():
                outputs = model(volumes)
                loss = loss_fn(outputs, masks)
            
            scaler.scale(loss).backward()
            scaler.unscale_(optimizer)
            torch.nn.utils.clip_grad_norm_(model.parameters(), max_norm=1.0)
            scaler.step(optimizer)
            scaler.update()
        else:
            outputs = model(volumes)
            loss = loss_fn(outputs, masks)
            loss.backward()
            torch.nn.utils.clip_grad_norm_(model.parameters(), max_norm=1.0)
            optimizer.step()
        
        total_loss += loss.item()
    
    avg_loss = total_loss / len(dataloader)
    return avg_loss
```

```python
def validate_epoch(model, dataloader, loss_fn, device):
    """
    Validate on validation set

        Dictionary with loss and metrics
    """
    model.eval()
    total_loss = 0.0
    all_dice = []
    all_recall = []
    all_precision = []
    all_f1 = []
    
    with torch.no_grad():
        for volumes, masks in tqdm(dataloader):
            volumes = volumes.to(device)

            loss = loss_fn(outputs, masks)

            all_precision.append(metrics['precision'])

    return {
        'loss': total_loss / len(dataloader),
        'dice': np.mean(all_dice),
```

```python
def save_checkpoint(model, optimizer, epoch, metrics, path):
    """
    Save model checkpoint
    
    Args:
        model: PyTorch model
        optimizer: Optimizer
        epoch: Current epoch
        metrics: Dictionary of metrics
        path: Save path
    """
    checkpoint = {
        'epoch': epoch,
        'model_state_dict': model.state_dict(),
        'optimizer_state_dict': optimizer.state_dict(),
        'metrics': metrics
    }
    torch.save(checkpoint, path)
    print(f"Checkpoint saved: {path}")
```

Function: load_checkpoint()

```python
def load_checkpoint(model, optimizer, path, device='cuda'):
    """
    Load model checkpoint
    
    Returns:
        Epoch number and metrics
    """
    checkpoint = torch.load(path, map_location=device)
    model.load_state_dict(checkpoint['model_state_dict'])
    if optimizer:
        optimizer.load_state_dict(checkpoint['optimizer_state_dict'])
    
    epoch = checkpoint['epoch']
    metrics = checkpoint['metrics']
    
    print(f"Loaded checkpoint from epoch {epoch}")
    return epoch, metrics
```

```python
from final_model import HybridMiniSwin2_5D_ResNet_CSRF
from inference import preprocess_volume, predict_volume, postprocess_mask

class ModelHandler:
    def __init__(self, checkpoint_path, device='cuda'):
        """Initialize model handler"""
        self.device = torch.device(device if torch.cuda.is_available() else 'cpu')
        self.model = self._load_model(checkpoint_path)
        self.loaded = True
    
    def _load_model(self, checkpoint_path):
        """Load model from checkpoint"""
        model = HybridMiniSwin2_5D_ResNet_CSRF()
        checkpoint = torch.load(checkpoint_path, map_location=self.device)
        model.load_state_dict(checkpoint['model_state_dict'])
        model.to(self.device)
        model.eval()
        return model
    
    def predict(self, flair_path, t1_path, t2_path):
        """Run inference on volume"""
        # Preprocess
        volume = preprocess_volume(flair_path, t1_path, t2_path)
        
        # Predict
        with torch.no_grad():
            mask = predict_volume(self.model, volume, self.device)
        
        # Post-process
        mask = postprocess_mask(mask)
        
        # Compute statistics
        lesion_volume = np.sum(mask) * 1.0  # mm³ (1mm isotropic)
        lesion_count = self._count_lesions(mask)
        
        # Save mask
        mask_url = self._save_mask(mask)
        
        return {
            'mask': mask,
            'lesion_volume': float(lesion_volume),
            'lesion_count': int(lesion_count),
            'mask_url': mask_url
        }
    
    def _count_lesions(self, mask):
```
